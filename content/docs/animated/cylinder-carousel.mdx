## Overview

Use Cylinder Carousel for visually browsable items where spatial exploration matters. Pointer drag, wheel input, momentum, and snapping share one continuous position model.

## Installation

<CliBlock commands={["cylinder-carousel"]} />

## Usage

```tsx
import { CylinderCarousel } from "@/components/animated/cylinder-carousel"

<CylinderCarousel visibleItems={5} variant="concave">
  {items.map((item) => <Card key={item.id} {...item} />)}
</CylinderCarousel>
```

## Accessibility

Do not use the carousel as the only route to essential content. Give interactive children accessible names and provide an equivalent list or navigation path when exact selection matters.

## API Reference

<ApiTable>
  <ApiRow name="children" type="ReactNode" required>Carousel items.</ApiRow>
  <ApiRow name="itemSize" type="number" default="200">Maximum square item size in pixels.</ApiRow>
  <ApiRow name="visibleItems" type="number" default="5">Item slots across the stage.</ApiRow>
  <ApiRow name="variant" type='"concave" | "convex"' default='"concave"'>Cylinder projection direction.</ApiRow>
  <ApiRow name="minScale" type="number" default="0.55">Smallest item scale.</ApiRow>
  <ApiRow name="snap" type="boolean" default="true">Settles on the nearest item.</ApiRow>
  <ApiRow name="autoRotate" type="boolean" default="false">Rotates until the user interacts.</ApiRow>
  <ApiRow name="onIndexChange" type="(index: number) => void">Runs when the settled item changes.</ApiRow>
</ApiTable>
